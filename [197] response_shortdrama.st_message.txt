// sdBack() walks THAT: previous trail URL, else the caller's fallback.
// Recording is idempotent breadcrumb-style: re-seeing a URL rewinds the
// trail to it, which absorbs Turbo restoration visits, double-records, and
// revisits (Home→A→Home→B backs to Home, not A). Identity is the URL the
// page was RECORDED under (__SD_NAV_SELF), so in-page replaceState churn
// (feed episode position, profile tab hashes, admin filters) never
// misaligns the walk. Overlay pops (sheets, search modal, lightbox) keep
// their own history.back() — they pop entries they pushed themselves.
(function () {
var KEY = 'sd_nav_trail';
function loadTrail() {
try { var t = JSON.parse(sessionStorage.getItem(KEY) || '[]'); return Array.isArray(t) ? t : []; }
catch (_) { return []; }
}
function saveTrail(t) { try { sessionStorage.setItem(KEY, JSON.stringify(t.slice(-40))); } catch (_) { } }
function here() { return location.pathname + location.search; }
function record() {
var u = here(), t = loadTrail(), i = t.lastIndexOf(u);
if (i !== -1) t.length = i + 1; else t.push(u);
saveTrail(t);
window.__SD_NAV_SELF = u;
}
window.sdBack = function (fallback) {
var t = loadTrail(), self = window.__SD_NAV_SELF || here(), i = t.lastIndexOf(self);
if (i !== -1) { t.length = i; saveTrail(t); } // drop self: a fallback landing must not ping-pong back
var prev = t.length ? t[t.length - 1] : null;
window.sdVisit(prev || fallback || window.sdUrl('/home'));
};
record();
if (!window.__SD_NAV_BOUND) { // shim re-runs per Turbo body swap; bind once
window.__SD_NAV_BOUND = 1;
// turbo:load covers every shell visit INCLUDING restorations (which do
// not re-execute body scripts); the inline record() above covers full
// loads and the no-Turbo feed layout. Re-records dedupe via the rewind.
document.addEventListener('turbo:load', record);
}
})();
</script>
